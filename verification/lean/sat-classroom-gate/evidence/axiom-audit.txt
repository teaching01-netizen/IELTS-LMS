  (∃ receipt, first.snd = Reply.created receipt ∧ second.snd = Reply.replayed receipt) ∧

SatClassroomGate.mixed_instance_configuration_is_not_global :
  let acceptedRequest := sampleRequest 2;
  let refusedRequest := sampleRequest 3;
  let store := Store.empty partiallyEnabled;
  let accepted := handle store 0 acceptedRequest;
  let refused := handle accepted.fst 1 refusedRequest;
  (∃ receipt, accepted.snd = Reply.created receipt) ∧
    refused.snd = Reply.classroomDisabled ∧ refused.fst.roomEffects acceptedRequest.key = 1
SatClassroomGate.clearing_generic_5xx_key_can_duplicate_rooms :
  let firstRequest := sampleRequest 3 Session.practice;
  let secondRequest := sampleRequest 4 Session.practice;
  let first := handle (Store.empty fun x => false) 0 firstRequest;
  let second := handle first.fst 0 secondRequest;
  first.fst.roomEffects firstRequest.key = 1 ∧
    second.fst.roomEffects firstRequest.key = 1 ∧
      second.fst.roomEffects secondRequest.key = 1 ∧ second.fst.totalRooms = 2
