 "testing"

func TestAuditUnassignedSATModuleMustNotBeActive(t *testing.T) {
 db,mock,err:=sqlmock.New();if err!=nil {t.Fatal(err)};defer db.Close()
 mock.ExpectQuery("SELECT m.id, s.section_key, COALESCE").WillReturnRows(sqlmock.NewRows([]string{"id","section_key","state"}).AddRow("unassigned-branch","reading-writing",""))
 owner,err:=(v2Resolver{}).Resolve(context.Background(),db,"attempt","future-question")
 if err==nil && owner.ModuleState=="active" {t.Fatal("unassigned SAT branch resolves as active and passes the V2 module write gate")}
 if err:=mock.ExpectationsWereMet();err!=nil {t.Fatal(err)}
}
